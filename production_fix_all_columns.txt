import sys; sys.path.append('.'); 
from app.main import app, db; 
from sqlalchemy import text; 
app.app_context().push(); 
conn = db.engine.connect(); 
print('🔧 Fixing production database...'); 
expected_cols = ['phone', 'is_active', 'email_verified', 'phone_verified', 'email_verification_token', 'phone_verification_code', 'phone_verification_expires', 'email_notifications', 'sms_notifications', 'push_notifications', 'oauth_provider', 'oauth_id', 'oauth_email', 'oauth_name', 'oauth_picture']; 
existing = [row[0] for row in conn.execute(text('SELECT column_name FROM information_schema.columns WHERE table_name = \\'users\\''))]; 
missing = [col for col in expected_cols if col not in existing]; 
print(f'Missing columns: {missing}'); 
for col in missing: 
    try: 
        if col == 'phone': conn.execute(text('ALTER TABLE users ADD COLUMN phone VARCHAR(20)')); 
        elif col == 'is_active': conn.execute(text('ALTER TABLE users ADD COLUMN is_active BOOLEAN DEFAULT TRUE')); 
        elif col == 'email_verified': conn.execute(text('ALTER TABLE users ADD COLUMN email_verified BOOLEAN DEFAULT FALSE'));

        elif col == 'email_verification_token': conn.execute(text('ALTER TABLE users ADD COLUMN email_verification_token VARCHAR(255)')); 
        elif col == 'phone_verification_code': conn.execute(text('ALTER TABLE users ADD COLUMN phone_verification_code VARCHAR(6)')); 
        elif col == 'phone_verification_expires': conn.execute(text('ALTER TABLE users ADD COLUMN phone_verification_expires TIMESTAMP')); 
        elif col == 'email_notifications': conn.execute(text('ALTER TABLE users ADD COLUMN email_notifications BOOLEAN DEFAULT TRUE')); 
        elif col == 'sms_notifications': conn.execute(text('ALTER TABLE users ADD COLUMN sms_notifications BOOLEAN DEFAULT FALSE')); 
        elif col == 'push_notifications': conn.execute(text('ALTER TABLE users ADD COLUMN push_notifications BOOLEAN DEFAULT TRUE'));

        elif col == 'oauth_id': conn.execute(text('ALTER TABLE users ADD COLUMN oauth_id VARCHAR(255)'));

        print(f'✅ Added: {col}'); 
    except Exception as e: 
        if 'already exists' in str(e).lower(): 
            print(f'ℹ️  Already exists: {col}'); 
        else: 
            print(f'❌ Error adding {col}: {e}'); 
conn.commit(); 
print('✅ All columns fixed');
